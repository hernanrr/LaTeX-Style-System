\begin{tealbox}
\textbf{Escala de evaluación.}
\begin{itemize}
  \item \textbf{Excelente}: cumple \emph{todos} los requisitos con rigor
    y precisión; el resultado es correcto y completo.
  \item \textbf{Satisfactorio}: cumple los requisitos esenciales; puede haber
    errores menores o aspectos incompletos que no afectan el resultado global.
  \item \textbf{En desarrollo}: cumple parcialmente; hay evidencia de
    comprensión del objetivo, pero los errores afectan la calidad del resultado.
  \item \textbf{Insuficiente}: no cumple los requisitos mínimos,
    o el entregable está ausente.
\end{itemize}
\end{tealbox}

\crithead{teal}%
  {C1 --- Serie de Máximos Anuales (AMS) \quad(\num{15}~pts)}%
  {Entregables E1, E2}
\nivel{15 pts}{AMS construida correctamente para Santo Domingo; aplica el
  criterio de exclusión de A1 documentando años excluidos; reporta $n$,
  $\bar{x}$, $s$, $CV$, $g_1$, mínimo y máximo; gráfico de barras con
  los eventos de 2023 y 2024 señalados; histograma con intervalos apropiados.}&
\nivel{11 pts}{AMS correcta; estadísticos presentes aunque incompletos ($CV$
  o $g_1$ ausentes); gráfico correcto; eventos no señalados explícitamente
  o histograma con intervalos inadecuados.}&
\nivel{8 pts}{AMS con error en la selección del máximo anual (e.g., usa
  promedio) o sin aplicar el criterio de exclusión; estadísticos incompletos.}&
\nivel{0 pts}{AMS ausente o con errores fundamentales que invalidan el
  análisis posterior; entregables ausentes.}&
15\\[3pt]

\crithead{teal}%
  {C2 --- Distribución Empírica (Weibull) \quad(\num{15}~pts)}%
  {Entregables E1, E3, E4}
\nivel{15 pts}{Valores ordenados de mayor a menor con rango $m$ correcto
  ($m=1$ para el mayor); $\hat{P}=m/(n+1)$; $T_m=(n+1)/m$; curva de
  excedencia (escala lineal en $P$) y curva de período de retorno (escala
  logarítmica en $T$); posiciones colocadas en ambas plantillas; interpolación
  limitada a $T\leq n+1$ y N/A más allá del registro.}&
\nivel{11 pts}{Cálculos correctos; una plantilla incompleta o error menor de
  escala; no se extrapola la distribución empírica.}&
\nivel{8 pts}{Usa fórmula de ploteo incorrecta (e.g., California $m/n$)
  o confunde excedencia con no-excedencia; o prolonga la curva empírica fuera
  del rango observado.}&
\nivel{0 pts}{Distribución empírica ausente; no se construye ninguna curva
  de frecuencia.}&
15\\[3pt]

\crithead{teal}%
  {C3 --- Ajuste Gumbel EVI \quad(\num{20}~pts)}%
  {Entregables E1, E3, E4}
\nivel{20 pts}{$\alpha_G$ y $u$ correctos por el método de los momentos;
  variable reducida $y$ calculada para cada observación; $F(x)$ y $1-F(x)$
  correctas; curva Gumbel superpuesta a la empírica; evaluación del ajuste
  con rangos de sobre/subestimación identificados; valores de diseño para
  todos los $T$ requeridos usando $y_T=-\ln[\ln(T/(T-1))]$ y
  $x_T=u+\alpha_G\,y_T$.}&
\nivel{15 pts}{Parámetros correctos; curvas presentes; evaluación superficial
  (sin rangos específicos); valores de diseño con errores menores.}&
\nivel{10 pts}{Error en $\alpha_G$ o $u$ que se propaga; curva no superpuesta
  a la empírica; sin evaluación del ajuste.}&
\nivel{0 pts}{Ajuste Gumbel ausente; confusión fundamental con la distribución
  o sus parámetros.}&
20\\[3pt]

\crithead{teal}%
  {C4 --- Ajuste SQRT-ETmax \quad(\num{15}~pts)}%
  {Entregables E1, E3, E4}
\nivel{15 pts}{CDF y log-verosimilitud SQRT-ETmax correctas; $k(\alpha)$ y
  $g(\alpha)$ implementados; raíz positiva con $|g|<10^{-6}$; $k,\alpha>0$;
  cuantiles obtenidos resolviendo la ecuación no lineal; CDF monótona; curva
  superpuesta y comparación explícita con Gumbel~EVI.}&
\nivel{11 pts}{Máxima verosimilitud y cuantiles esencialmente correctos;
  curva presente; error menor de tolerancia, verificación o comparación.}&
\nivel{8 pts}{Modelo SQRT-ETmax identificado, pero con errores en la
  log-verosimilitud, la raíz numérica o la inversión de cuantiles.}&
\nivel{0 pts}{Ajuste SQRT-ETmax ausente o sustituido por Gumbel aplicado a
  $\sqrt{x}$.}&
15\\[3pt]

\crithead{teal}%
  {C5 --- Tabla de Precipitaciones de Diseño \quad(\num{15}~pts)}%
  {Entregable E5}
\nivel{15 pts}{Tabla completa para los nueve $T$ requeridos (2 a 1000 años)
  con los tres métodos; N/A empírico cuando $T>n+1$; extrapolaciones
  paramétricas de 200, 500 y 1000 años marcadas como de alta incertidumbre;
  valores consistentes con las curvas y unidades indicadas.}&
\nivel{11 pts}{Tabla con los nueve $T$; al menos dos métodos correctos;
  incertidumbre o límite empírico no señalados; unidades presentes.}&
\nivel{8 pts}{Tabla incompleta; valores inconsistentes con las curvas;
  extrapolación confundida con interpolación.}&
\nivel{0 pts}{Tabla ausente o con un solo método; no permite tomar
  decisiones de diseño.}&
15\\[3pt]

\crithead{teal}%
  {C6 --- Análisis de Eventos Extremos 2023--2024 \quad(\num{10}~pts)}%
  {Entregable E6}
\nivel{10 pts}{Máximo diario identificado para los tres eventos (nov.~2023,
  nov.~2024, dic.~2024); ranking histórico correcto; período de retorno por
  los tres métodos; tabla comparativa completa; eventos marcados en las
  curvas; discusión sobre rareza estadística y coherencia entre métodos.}&
\nivel{7 pts}{Los tres eventos identificados; período de retorno con al menos
  dos métodos; tabla presente aunque incompleta; discusión superficial.}&
\nivel{5 pts}{Solo uno o dos eventos analizados; período de retorno con un
  solo método o con errores; sin discusión.}&
\nivel{0 pts}{Análisis de eventos extremos ausente.}&
10\\[3pt]

\crithead{teal}%
  {C7 --- Preguntas de Análisis (P1--P5) \quad(\num{10}~pts, \num{2}~pts c/u)}%
  {Entregable E7}
\nivel{2 pts c/u}{Respuesta correcta y completa; incluye cálculo explícito
  donde aplique (P1 en particular); usa resultados propios como evidencia.}&
\nivel{1.5 pts c/u}{Correcta en lo esencial; cálculos con errores menores
  o sin conexión con resultados propios.}&
\nivel{1 pt c/u}{Parcialmente correcta; errores conceptuales o falta de
  justificación cuantitativa.}&
\nivel{0 pts c/u}{Ausente, incorrecta o sin ninguna justificación.}&
10\\[3pt]
