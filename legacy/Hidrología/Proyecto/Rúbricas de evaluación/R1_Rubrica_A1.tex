\begin{infobox}
\textbf{Escala de evaluación.}
\begin{itemize}
  \item \textbf{Excelente}: cumple \emph{todos} los requisitos con rigor
    y precisión; el resultado es correcto y completo.
  \item \textbf{Satisfactorio}: cumple los requisitos esenciales; puede haber
    errores menores o aspectos incompletos que no afectan el resultado global.
  \item \textbf{En desarrollo}: cumple parcialmente; hay evidencia de
    comprensión del objetivo, pero los errores afectan la calidad del resultado.
  \item \textbf{Insuficiente}: no cumple los requisitos mínimos,
    o el entregable está ausente.
\end{itemize}
\end{infobox}

\crithead{medblue}%
  {C1 --- Lectura y Serialización \quad(\num{20}~pts)}%
  {Entregables E1, E2 (parcial)}
\nivel{20 pts}{Lee las 6 hojas; localiza bloques anuales con el texto
  \texttt{"DATOS DIARIOS"}; trata guión $\to$ NaN,
  INAP $\to$ \qty{0}{mm} y coma decimal $\to$ punto sin excepción;
  no genera fechas inválidas; verifica totales contra fila \texttt{TOTAL}
  con diferencias $\leq\qty{0.5}{mm}$; CSVs con columnas
  fecha, estación, precipitación.}&
\nivel{15 pts}{Lee y serializa $\geq4$ estaciones; trata al menos 2
  de los 3 valores especiales; verifica algunos totales; CSVs utilizables.}&
\nivel{10 pts}{Serializa al menos una estación; errores en valores especiales
  o estructura; verificación ausente o con errores sistemáticos.}&
\nivel{0 pts}{No logra serializar datos utilizables; errores fundamentales;
  entregable ausente.}&
20\\[3pt]

\crithead{medblue}%
  {C2 --- Mapa de Calidad de Datos \quad(\num{10}~pts)}%
  {Entregable E2}
\nivel{10 pts}{Tabla año $\times$ estación completa para las 6 estaciones;
  distingue días inexistentes del calendario de días genuinamente faltantes;
  coloreada por los 4 rangos de completitud; criterio de exclusión definido,
  documentado y justificado con evidencia de los datos.}&
\nivel{7 pts}{Tabla para $\geq4$ estaciones; criterio de exclusión indicado
  sin justificación explícita; puede confundir días inexistentes con
  faltantes en algunos casos.}&
\nivel{5 pts}{Tabla incompleta ($<4$ estaciones) o sin colorear; criterio
  de exclusión no definido.}&
\nivel{0 pts}{Mapa de calidad ausente o fundamentalmente incorrecto.}&
10\\[3pt]

\crithead{medblue}%
  {C3 --- Curva de Doble Masa \quad(\num{20}~pts)}%
  {Entregable E3}
\nivel{20 pts}{Acumulados históricos correctos para Santo Domingo y Las Américas
  como referencia fija; años concurrentes filtrados con el mismo criterio;
  gráfico con puntos etiquetados; quiebres identificados o ausencia justificada;
  segmentos con al menos ocho años; corrección aplicada solo si corresponde;
  comparación original vs.\ corregida; discusión de causas y de la estabilidad
  de la referencia.}&
\nivel{15 pts}{Curva correcta; quiebres identificados; discusión presente
  aunque incompleta; corrección ausente o con error menor que no afecta
  la interpretación global.}&
\nivel{10 pts}{Errores en los acumulados o la referencia; interpretación
  superficial; sin corrección ni discusión.}&
\nivel{0 pts}{Curva ausente o usa valores anuales directos en lugar de
  acumulados históricos.}&
20\\[3pt]

\crithead{medblue}%
  {C4 --- Estimación de Datos Faltantes \quad(\num{15}~pts)}%
  {Entregable E4}
\nivel{15 pts}{Normales anuales con al menos diez años válidos; razón normal
  aplicada solo a faltantes diarios aislados de La Victoria con las tres vecinas
  observadas; datos originales preservados; columnas de estimación y trazabilidad
  presentes; meses/años completos y secuencias largas permanecen sin rellenar;
  resumen temporal reportado.}&
\nivel{11 pts}{Método y límites correctos; tabla presente; puede haber un error
  menor en las normales, las banderas o la documentación.}&
\nivel{8 pts}{Estimaciones realizadas, pero con errores en la fórmula,
  sobrescritura parcial de originales o aplicación a lagunas demasiado largas.}&
\nivel{0 pts}{Estimación ausente, datos originales sobrescritos o lagunas
  extensas reconstruidas sin justificación.}&
15\\[3pt]

\crithead{medblue}%
  {C5 --- Análisis Estadístico Exploratorio \quad(\num{20}~pts)}%
  {Entregables E5, E6 (parcial)}
\nivel{20 pts}{Tabla de estadísticas (media, $s$, CV, mín., máx.,
  percentiles) para todas las estaciones; climatología mensual con
  $\pm1\sigma$ para Santo Domingo y Las Américas en un mismo panel; gráfico
  de \textbf{barras} de la serie diaria con justificación escrita; gráfico
  de series anuales con media histórica; discusión del régimen pluviométrico
  (unimodal/bimodal) con referencia a sistemas meteorológicos de la región.}&
\nivel{15 pts}{Mayoría de gráficos correctos; tabla presente aunque incompleta;
  discusión del régimen presente; puede faltar justificación del tipo de
  gráfico o las barras de error.}&
\nivel{10 pts}{Gráficos con errores conceptuales (e.g., línea para datos
  diarios) o incompletos; tabla ausente o parcial; poca o ninguna discusión.}&
\nivel{0 pts}{Análisis exploratorio ausente; gráficos ilegibles o con
  errores conceptuales fundamentales.}&
20\\[3pt]

\crithead{medblue}%
  {C6 --- Preguntas de Análisis (P1--P5) \quad(\num{15}~pts, \num{3}~pts c/u)}%
  {Entregable E6}
\nivel{3 pts c/u}{Respuesta completa y correcta; usa datos concretos del
  propio análisis; demuestra comprensión del concepto más allá de su
  definición.}&
\nivel{2 pts c/u}{Correcta en lo esencial pero incompleta; justificación
  parcial o sin referencia a los resultados propios.}&
\nivel{1 pt c/u}{Parcialmente correcta; errores menores o falta de
  justificación cuantitativa.}&
\nivel{0 pts c/u}{Ausente, incorrecta o sin ninguna justificación.}&
15\\[3pt]
