\textbf{Escala de evaluación.}
\begin{itemize}
  \item \textbf{Excelente}: cumple \emph{todos} los requisitos con rigor
    y precisión; el resultado es correcto y completo.
  \item \textbf{Satisfactorio}: cumple los requisitos esenciales; puede haber
    errores menores o aspectos incompletos que no afectan el resultado global.
  \item \textbf{En desarrollo}: cumple parcialmente; hay evidencia de
    comprensión del objetivo, pero los errores afectan la calidad del resultado.
  \item \textbf{Insuficiente}: no cumple los requisitos mínimos,
    o el entregable está ausente.
\end{itemize}

\crithead{olive}%
  {C1 --- Configuración SIG y Capa de Estaciones \quad(\num{15}~pts)}%
  {Entregables E1, E2 (parcial)}
\nivel{15 pts}{Proyecto \qgis{} organizado en grupos (Estaciones, Límites,
  Thiessen, Isoyetas, MDE) con rutas relativas; SRC del proyecto EPSG:32619;
  capa de cinco puntos EPSG:4326 con los 7 atributos requeridos
  (incluidos \texttt{periodo\_media} y las dos precipitaciones); coordenadas
  DMS convertidas y longitudes negativas; simbología graduada por la media del período;
  etiquetas de nombre visibles; posición verificada sobre mapa base.}&
\nivel{11 pts}{Proyecto organizado; atributos esenciales presentes; SRC
  correcto; DMS convertidas; simbología presente aunque no graduada.}&
\nivel{8 pts}{$\geq1$ estación mal ubicada, DMS sin convertir, o SRC
  incorrecto; atributos incompletos.}&
\nivel{0 pts}{Capa de puntos ausente o con errores fundamentales (coordenadas
  invertidas, estaciones en posición incorrecta); proyecto no entregado.}&
15\\[3pt]

\crithead{olive}%
  {C2 --- Composición Cartográfica: Mapa de Estaciones \quad(\num{10}~pts)}%
  {Entregable E2}
\nivel{10 pts}{Composición exportada a $\geq\qty{200}{dpi}$; contiene: título,
  escala gráfica, flecha de norte, leyenda con simbología de $\bar{P}$ del
  período disponible y rangos, créditos de fuentes; capas visibles: cinco
  estaciones, límite del DN y mapa base o MDE; períodos indicados y etiquetas legibles.}&
\nivel{7 pts}{Composición exportada; al menos título, escala y norte; capas
  principales visibles; puede faltar leyenda de $\bar{P}$ o créditos.}&
\nivel{5 pts}{Composición con $\geq3$ elementos cartográficos faltantes;
  resolución baja; contexto geográfico deficiente.}&
\nivel{0 pts}{Composición ausente; solo captura de pantalla sin elementos
  cartográficos.}&
10\\[3pt]

\crithead{olive}%
  {C3 --- Construcción de Polígonos de Thiessen \quad(\num{15}~pts)}%
  {Entregables E1, E3 (parcial)}
\nivel{15 pts}{Polígonos de Voronoi generados con buffer suficiente para
  cubrir el DN; recortados correctamente al contorno del DN; áreas
  calculadas en km² con la calculadora de campos (SRC métrico); fracciones
  $w_i$ calculadas y sumadas a 1; join con datos de precipitación correcto;
  mapa con simbología por precipitación de nov.~2023 y estaciones
  etiquetadas.}&
\nivel{11 pts}{Polígonos generados y recortados; áreas calculadas; join
  realizado; mapa presente; error menor en el recorte o las fracciones.}&
\nivel{8 pts}{Polígonos sin recortar al DN; áreas en unidades incorrectas
  (no km²); join ausente o con errores.}&
\nivel{0 pts}{Polígonos de Thiessen no generados; división arbitraria o
  ningún análisis de ponderación espacial.}&
15\\[3pt]

\crithead{olive}%
  {C4 --- Precipitación Areal por Thiessen \quad(\num{20}~pts)}%
  {Entregable E4}
\nivel{20 pts}{Tabla completa para nov.~2023 \emph{y} para la media anual del
  período disponible: columnas $A_i$ (km²), $w_i$, $P_i$ (mm), $w_iP_i$ (mm)
  y \texttt{periodo\_media}; usa los mismos polígonos y pesos en ambos cálculos;
  $\bar{P}=\sum w_iP_i$ correcto; comparado con promedio aritmético
  (diferencia en mm y en \%); discusión sobre qué estación domina y si
  es razonable geográficamente; advierte que las medias no forman una normal común.}&
\nivel{15 pts}{Tabla para nov.~2023 correcta; $\bar{P}$ calculado; comparación
  con promedio aritmético presente; tabla de medias del período ausente o
  incompleta; sin argumentación geográfica.}&
\nivel{10 pts}{Errores en $w_i$ (no suma a 1) o $\bar{P}$ calculado como
  promedio simple sin ponderar; comparación ausente; sin discusión.}&
\nivel{0 pts}{Cálculo por Thiessen ausente; solo se reporta el promedio
  aritmético.}&
20\\[3pt]

\crithead{olive}%
  {C5 --- Interpolación IDW y Extracción de Isoyetas \quad(\num{15}~pts)}%
  {Entregables E1, E5 (parcial)}
\nivel{15 pts}{IDW configurado correctamente (extensión cubre DN + buffer,
  celda $\leq\qty{500}{m}$, potencia 2); ráster plausible (valores dentro del
  rango observado); isoyetas extraídas con intervalos de \qty{50}{mm} y
  recortadas al DN; mapa con ráster IDW de fondo semitransparente, isoyetas
  etiquetadas en mm y elementos cartográficos completos.}&
\nivel{11 pts}{IDW correcto; ráster plausible; isoyetas extraídas y recortadas;
  mapa con los elementos esenciales; puede faltar el ráster de fondo o las
  etiquetas.}&
\nivel{8 pts}{IDW con parámetros inadecuados; ráster de baja calidad;
  isoyetas no recortadas al DN o con intervalos inadecuados.}&
\nivel{0 pts}{Interpolación no realizada; isoyetas ausentes o trazadas
  manualmente sin \qgis{}.}&
15\\[3pt]

\crithead{olive}%
  {C6 --- Precipitación Areal por Isoyetas \quad(\num{15}~pts)}%
  {Entregable E6}
\nivel{15 pts}{Tabla completa: intervalo de isoyetas, $\bar{P}_j$ (mm),
  $A_j$ (km²), $A_j\bar{P}_j$; áreas parciales calculadas en \qgis{} (no
  estimadas visualmente); $\bar{P}$ calculado por la fórmula de isoyetas;
  comparación con Thiessen en mm y en \%; discusión sobre cuál método es
  más apropiado dado el número de estaciones y las limitaciones del IDW.}&
\nivel{11 pts}{Tabla con columnas esenciales; $\bar{P}$ correcto; comparación
  con Thiessen presente; discusión sin cuantificar la diferencia.}&
\nivel{8 pts}{Áreas parciales estimadas visualmente o $\bar{P}$ como promedio
  simple de $\bar{P}_j$ sin ponderar por $A_j$; comparación ausente.}&
\nivel{0 pts}{Cálculo por isoyetas ausente.}&
15\\[3pt]

\crithead{olive}%
  {C7 --- Preguntas de Análisis (P1--P5) \quad(\num{10}~pts, \num{2}~pts c/u)}%
  {Entregable E7}
\nivel{2 pts c/u}{Respuesta correcta y completa; usa resultados propios
  (áreas de polígonos, valores de $\bar{P}$, diferencias entre métodos);
  argumentación geográfica presente; capturas en P4 y P5 donde aplique.}&
\nivel{1.5 pts c/u}{Correcta en lo esencial; referencia a resultados propios
  presente aunque imprecisa.}&
\nivel{1 pt c/u}{Parcialmente correcta; responde genéricamente sin usar
  los resultados del propio análisis.}&
\nivel{0 pts c/u}{Ausente o incorrecta; respuesta copiada sin adaptación.}&
10\\[3pt]
